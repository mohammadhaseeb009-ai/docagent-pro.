Apps Script — doGet (Live Fetch)
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var rows = ss.getActiveSheet().getDataRange().getValues();
  var headers = rows[0];
  var data = rows.slice(1).map(function(r){
    var obj={}; headers.forEach(function(h,i){obj[h]=r[i];}); return obj;
  });
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
Apps Script — doPost (Form Submit)
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([data.timestamp, data.project,
    data.mandal, data.sector, data.awcName,
    data.awcId, data.remarks]);
  return ContentService.createTextOutput('OK');
}
Setup: Open Google Sheet → Extensions → Apps Script → paste both functions → Deploy as Web App → Copy URL → paste in Setup above.
